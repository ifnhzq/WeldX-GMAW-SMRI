# Thermal Analysis of GMAW Welding Process on Mild Steel (AISI 1018 / S235)
# =============================================================
#
# This notebook implements Rosenthal's 3D Moving Point Heat Source Model with:
#   - Multiple simultaneous sources at 5mm, 10mm, and 20mm from weld path
#   - Monte Carlo simulation over 100 samples of process variability
#   - Physical constraints (melting point capping, singularity avoidance)
#
# Rosenthal's Equation (3D):
#   T(x,y,z) - T0 = Q / (2*pi*k*R) * exp(-v*(x+R) / (2*alpha))
#
# Where:
#   Q     = Net heat input power [W]
#   k     = Thermal conductivity [W/m·K]
#   R     = sqrt(x² + y² + z²)  [m]
#   v     = Welding speed [m/s]
#   alpha = Thermal diffusivity [m²/s]
#   T0    = Preheat temperature [°C]
#
# ⚠ Prerequisites: Run Tab 1, Tab 2, and Tab 3 first.
# =============================================================

# Evaluated in the quasi-stationary frame moving with the heat source.
# x is the distance behind the source (negative = ahead, positive = behind).
 
def rosenthal_3d(Q, v, k, alpha, T0, x, y, z, T_melting, R_min=3e-3):
    """
    Rosenthal 3D Moving Point Heat Source (Butt Joint — Mild Steel GMAW)
 
    Parameters
    ----------
    Q         : Net power [W]
    v         : Welding speed [m/s]
    k         : Thermal conductivity [W/m/K]
    alpha     : Thermal diffusivity [m²/s]
    T0        : Initial temperature [°C]
    x, y, z   : Coordinates in moving frame [m]
    T_melting : Melting point [°C]
    R_min     : Singularity limiter [m]
 
    Returns
    -------
    T : Temperature field [°C]
    """
    R = np.sqrt(x**2 + y**2 + z**2)
    R = np.maximum(R, R_min)            # avoid singularity at heat source
 
    # Rosenthal solution (steady moving source)
    T_C = T0 + (Q / (2.0 * np.pi * k * R)) * \
          np.exp(-v * (x + R) / (2.0 * alpha))
 
    # Physical constraint: cap at melting point
    T_C = np.minimum(T_C, T_melting)
 
    return T_C
 
print('[Tab 4] ✓ Rosenthal 3D model (Butt Joint Mild Steel GMAW) defined.')

import warnings
warnings.filterwarnings('ignore')
 
print(f'[Tab 4] Running {N} Monte Carlo simulations (Mild Steel GMAW Butt Joint)...')
print(f'        Sensors   : {sensor_labels}')
print(f'        Time steps: {len(t_array)}')
 
thermal_results   = {}
peak_temperatures = {}
 
for label, d_m in zip(sensor_labels, sensor_distances_m):
    T_mc = np.zeros((N, len(t_array)))   # [°C]
 
    for i in range(N):
        # Monte Carlo inputs
        Q_i     = mc['net_power'][i]       # [W]
        v_i     = mc['speed'][i] * 1e-3    # [m/s]  — speed stored in mm/s
        noise_i = mc['sensor_noise'][i]    # [°C]
 
        # Butt joint moving frame:
        # x = -v*t  (heat source moving along weld seam)
        x_arr = -v_i * t_array    # [m]
        y_arr = d_m               # [m]
        z_arr = 0.0               # [m] (surface)
 
        # Rosenthal solution
        T_i = rosenthal_3d(
            Q         = Q_i,
            v         = v_i,
            k         = k,
            alpha     = alpha,
            T0        = T0,
            x         = x_arr,
            y         = y_arr,
            z         = z_arr,
            T_melting = T_mp,
        )
 
        # Add measurement noise
        T_mc[i, :] = T_i + noise_i
 
    thermal_results[label]   = T_mc
    peak_temperatures[label] = T_mc.max(axis=1)
 
    T_mean = peak_temperatures[label].mean()
    T_std  = peak_temperatures[label].std()
    print(f'  {label}: Peak T = {T_mean:.1f} ± {T_std:.1f} °C')
 
print('\n[Tab 4] ✓ Monte Carlo Rosenthal simulation complete (Butt Joint Mild Steel GMAW).')
