# Monte Carlo Sampling using Normal (Gaussian) distributions
# CoV ~ 4% for voltage/current, ~4% for speed
 
# Voltage: mean=24V, std=1.0V (±4.2%)
mc_voltage = rng.normal(loc=V_nom, scale=1.0, size=N_SAMPLES)       # [V]
 
# Current: mean=200A, std=8A (±4%)
mc_current = rng.normal(loc=I_nom, scale=8.0, size=N_SAMPLES)       # [A]
 
# Travel Speed: mean=5.0 mm/s, std=0.2 mm/s (±4%)
mc_speed = rng.normal(loc=v_nom, scale=0.2, size=N_SAMPLES)         # [mm/s]
 
# Sensor Noise: mean=0°C, std=3°C (temperature sensor noise)
mc_sensor_noise = rng.normal(loc=0.0, scale=3.0, size=N_SAMPLES)    # [°C]
 
# Clip unphysical values
mc_voltage = np.clip(mc_voltage, 15.0, 35.0)
mc_current = np.clip(mc_current, 150.0, 280.0)
mc_speed   = np.clip(mc_speed,   1.0,   12.0)
 
# Derived: Net Power per sample [W] and Heat Input [J/m]
efficiency_nom  = float(core['process']['efficiency'].m)
mc_net_power    = mc_voltage * mc_current * efficiency_nom   # [W]
mc_heat_input   = mc_net_power / (mc_speed * 1e-3)           # [J/m]
 
print('\n=== Monte Carlo Statistics ===')
print(f'  Voltage      : mean={mc_voltage.mean():.2f} V,    std={mc_voltage.std():.3f} V')
print(f'  Current      : mean={mc_current.mean():.2f} A,   std={mc_current.std():.3f} A')
print(f'  Travel Speed : mean={mc_speed.mean():.4f} mm/s, std={mc_speed.std():.4f} mm/s')
print(f'  Sensor Noise : mean={mc_sensor_noise.mean():.3f} °C,  std={mc_sensor_noise.std():.3f} °C')
print(f'  Net Power    : mean={mc_net_power.mean():.1f} W,   std={mc_net_power.std():.2f} W')
print(f'  Heat Input   : mean={mc_heat_input.mean():.0f} J/m, std={mc_heat_input.std():.0f} J/m')

fig = plt.figure(figsize=(14, 10))
fig.suptitle(
    'Monte Carlo Parameter Distributions (N=100)\nGMAW Welding on Mild Steel AISI 1018 / S235 (Butt Joint)',
    fontsize=14, fontweight='bold'
)
 
gs = gridspec.GridSpec(2, 3, figure=fig, hspace=0.4, wspace=0.35)
 
datasets = [
    (mc_voltage,      'Voltage',      'V',    '#1f77b4'),   # blue
    (mc_current,      'Current',      'A',    '#d62728'),   # red
    (mc_speed,        'Travel Speed', 'mm/s', '#2ca02c'),   # green
    (mc_sensor_noise, 'Sensor Noise', '°C',   '#ff7f0e'),   # orange
    (mc_net_power,    'Net Power',    'W',    '#9467bd'),   # purple
    (mc_heat_input,   'Heat Input',   'J/m',  '#8c564b'),   # brown
]
 
for idx, (data, label, unit, color) in enumerate(datasets):
    ax = fig.add_subplot(gs[idx // 3, idx % 3])
    ax.hist(data, bins=15, color=color, alpha=0.80, edgecolor='black', linewidth=0.7)
    ax.axvline(data.mean(), color='black', linestyle='--', linewidth=1.5,
               label=f'Mean={data.mean():.2f}')
    ax.set_title(f'{label}', fontweight='bold')
    ax.set_xlabel(f'{label} [{unit}]')
    ax.set_ylabel('Frequency')
    ax.legend(fontsize=8)
    ax.grid(True, alpha=0.25)
 
plt.savefig('mc_distributions_gmaw.png', dpi=120, bbox_inches='tight')
plt.show()
print('[Tab 2] ✓ Distribution plot saved as mc_distributions_gmaw.png')
