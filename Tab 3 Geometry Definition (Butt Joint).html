# Butt Joint Coordinate System:
#   X-axis  → along weld travel direction
#   Y-axis  → distance from weld centerline (left/right)
#   Z-axis  → through thickness of plates
#
# Weld path: straight line along X-axis (butt joint seam)
 
weld_path_start = np.array([-plate_length/2, 0.0, 0.0])   # [m]
weld_path_end   = np.array([ plate_length/2, 0.0, 0.0])   # [m]
 
# Weld speed from core file
v_weld       = float(core['process']['speed'].to('m/s').m)   # [m/s]
weld_duration = plate_length / v_weld                         # [s]
 
# Time array for weld simulation
n_time_steps = 100
t_array      = np.linspace(0, weld_duration, n_time_steps)   # [s]
 
# Heat source position as function of time (moving along seam)
x_source = -plate_length/2 + v_weld * t_array                # [m]
 
# Sensor Positions — at X=0 (mid-length), Y = sensor distance, Z = surface
sensor_positions = []
for d, label in zip(sensor_distances, sensor_labels):
    pos = {
        'label': label,
        'x': 0.0,        # Mid-length [m]
        'y': float(d),   # Distance from weld centerline [m]
        'z': 0.0,        # Surface [m]
    }
    sensor_positions.append(pos)
    print(f'  Sensor {label}: y = {d*1000:.1f} mm from weld centerline')
 
print(f'\nWeld duration: {weld_duration:.2f} s at {v_weld*1000:.1f} mm/s')

import matplotlib.pyplot as plt

# ============================================================
# Figure Setup
# ============================================================
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(16, 6))

fig.suptitle(
    'Butt Joint Geometry & Thermal Sensor Configuration\n'
    'GMAW Welding on Mild Steel AISI 1018',
    fontsize=14,
    fontweight='bold'
)

# ============================================================
# TOP VIEW (XY PLANE)
# ============================================================
ax1.set_title('Top View (XY Plane)', fontweight='bold')

half_L = plate_length / 2 * 1000      # mm
half_W = plate_width / 2 * 1000       # mm

# ------------------------------------------------------------
# Plate
# ------------------------------------------------------------
plate_rect = plt.Rectangle(
    (-half_L, -half_W),
    plate_length * 1000,
    plate_width * 1000,
    fill=True,
    facecolor='lightgray',
    edgecolor='black',
    linewidth=2,
    label='Base Plate'
)

ax1.add_patch(plate_rect)

# ------------------------------------------------------------
# Weld Seam
# ------------------------------------------------------------
ax1.plot(
    [-half_L, half_L],
    [0, 0],
    'r-',
    linewidth=3,
    label='Weld Seam',
    zorder=5
)

ax1.plot(
    0,
    0,
    'r*',
    markersize=15,
    label='Weld Center',
    zorder=6
)

# ------------------------------------------------------------
# Welding Direction
# ------------------------------------------------------------
ax1.annotate(
    '',
    xy=(half_L * 0.70, 0),
    xytext=(half_L * 0.35, 0),
    arrowprops=dict(
        arrowstyle='->',
        color='red',
        lw=2.5
    )
)

ax1.text(
    half_L * 0.52,
    5,
    'Welding Direction',
    color='red',
    fontsize=9,
    ha='center'
)

# ------------------------------------------------------------
# Thermal Measurement Points
# ------------------------------------------------------------
sensor_colors = ['gold', 'darkorange', 'saddlebrown']

for sp, color in zip(sensor_positions, sensor_colors):

    x_mm = sp['x'] * 1000
    y_mm = sp['y'] * 1000

    # Sensor marker
    ax1.plot(
        x_mm,
        y_mm,
        'D',
        color=color,
        markersize=12,
        zorder=8
    )

    # Sensor label
    ax1.annotate(
        f"{sp['label']}\n({x_mm:.0f}, {y_mm:.0f}) mm",
        xy=(x_mm, y_mm),
        xytext=(10, 10),
        textcoords='offset points',
        fontsize=8,
        fontweight='bold'
    )

    # Vertical guide line
    ax1.plot(
        [x_mm, x_mm],
        [0, y_mm],
        ':',
        color=color,
        linewidth=1.5
    )

    # Dimension arrow
    ax1.annotate(
        '',
        xy=(0, y_mm),
        xytext=(0, 0),
        arrowprops=dict(
            arrowstyle='<->',
            linestyle='--',
            color=color,
            lw=1.5
        )
    )

    ax1.text(
        15,
        y_mm/2,
        f'{y_mm:.0f} mm',
        color=color,
        fontsize=8,
        fontweight='bold'
    )

# ------------------------------------------------------------
# Axis Settings
# ------------------------------------------------------------
ax1.set_xlabel('X — Welding Direction [mm]')
ax1.set_ylabel('Y — Distance from Weld Centerline [mm]')

# Wider than plate

ax1.set_ylim(-half_W * 1.5, half_W * 1.5)

geometry_extended = dict(core['geometry'])   # copy existing
 
geometry_extended['weld_path_start'] = Q_(weld_path_start * 1000, 'mm')
geometry_extended['weld_path_end']   = Q_(weld_path_end   * 1000, 'mm')
geometry_extended['weld_duration']   = Q_(weld_duration, 's')
geometry_extended['sensor_positions'] = [
    {
        'label': sp['label'],
        'x_mm' : sp['x'] * 1000,
        'y_mm' : sp['y'] * 1000,
        'z_mm' : sp['z'] * 1000,
    }
    for sp in sensor_positions
]
 
# Write updated geometry back to WeldX file
with WeldxFile(CORE_FILE, mode='rw') as wx:
    wx['geometry'] = geometry_extended
